export function formatTimestamp(iso?: string | null): string {
  if (!iso) return '—';
  const d = new Date(iso);
  if (Number.isNaN(d.getTime())) return iso;
  return d.toLocaleString('en-ZW', {
    day: '2-digit',
    month: 'short',
    year: 'numeric',
    hour: '2-digit',
    minute: '2-digit',
  });
}

export interface SmartButtonProps {
  label: string;
  onPress: () => void;
  variant?: 'primary' | 'secondary' | 'danger' | 'ghost';
  disabled?: boolean;
  loading?: boolean;
  style?: ViewStyle;
  accessibilityLabel?: string;
}

export const SmartButton: React.FC<SmartButtonProps> = ({
  label,
  onPress,
  variant = 'primary',
  disabled = false,
  loading = false,
  style,
  accessibilityLabel,
}) => {
  const isDisabled = disabled || loading;

  const containerVariantStyle: ViewStyle =
    variant === 'primary'
      ? { backgroundColor: COLORS.primary, borderColor: COLORS.primary }
      : variant === 'secondary'
      ? { backgroundColor: COLORS.surface, borderColor: COLORS.primary }
      : variant === 'danger'
      ? { backgroundColor: COLORS.danger, borderColor: COLORS.danger }
      : { backgroundColor: 'transparent', borderColor: 'transparent' };

  const textVariantStyle: TextStyle =
    variant === 'primary' || variant === 'danger'
      ? { color: COLORS.white }
      : variant === 'secondary'
      ? { color: COLORS.primary }
      : { color: COLORS.textMuted };

  return (
    <Pressable
      accessibilityRole="button"
      accessibilityLabel={accessibilityLabel || label}
      accessibilityState={{ disabled: isDisabled, busy: loading }}
      disabled={isDisabled}
      onPress={onPress}
      style={({ pressed }) => [
        styles.buttonBase,
        containerVariantStyle,
        isDisabled && styles.buttonDisabled,
        pressed && !isDisabled && styles.pressedOpacity,
        style,
      ]}
    >
      {loading ? (
        <ActivityIndicator
          size="small"
          color={
            variant === 'primary' || variant === 'danger'
              ? COLORS.white
              : COLORS.primary
          }
        />
      ) : null}
      <Text style={[styles.buttonText, textVariantStyle]}>{label}</Text>
    </Pressable>
  );
};

export interface OptionItem<T extends string> {
  value: T;
  label: string;
}

export interface OptionPillGroupProps<T extends string> {
  label: string;
  required?: boolean;
  options: OptionItem<T>[];
  selectedValue: T | '';
  onSelect: (value: T) => void;
  errorText?: string;
  columns?: 1 | 2;
}

export function OptionPillGroup<T extends string>({
  label,
  required,
  options,
  selectedValue,
  onSelect,
  errorText,
  columns = 2,
}: OptionPillGroupProps<T>) {
  return (
    <View style={styles.fieldContainer}>
      <View style={styles.fieldHeaderRow}>
        <Text style={styles.fieldLabel}>
          {label}
          {required ? <Text style={styles.requiredMark}> *</Text> : null}
        </Text>
      </View>
      <View style={styles.pillGrid}>
        {options.map((opt) => {
          const selected = selectedValue === opt.value;
          return (
            <Pressable
              key={opt.value}
              accessibilityRole="radio"
              accessibilityState={{ checked: selected }}
              onPress={() => onSelect(opt.value)}
              style={({ pressed }) => [
                styles.optionPill,
                columns === 1 ? styles.optionPillFull : styles.optionPillHalf,
                selected && styles.optionPillSelected,
                pressed && styles.pressedOpacity,
              ]}
            >
              <Text
                style={[
                  styles.optionPillText,
                  selected && styles.optionPillTextSelected,
                ]}
              >
                {opt.label}
              </Text>
              <Text
                style={[
                  styles.optionCheckMark,
                  selected && styles.optionPillTextSelected,
                ]}
              >
                {selected ? '●' : '○'}
              </Text>
            </Pressable>
          );
        })}
      </View>
      {errorText ? <Text style={styles.errorText}>{errorText}</Text> : null}
    </View>
  );
}

export const ReportIncidentScreen: React.FC<ReportIncidentScreenProps> = ({
  siteName,
  shiftStartedAt,
  isOnline,
  pendingCount = 0,
  initialEventAt,
  isSubmitting = false,
  storageError = null,
  onSubmitIncident,
  onCancel,
}) => {
  const [category, setCategory] = useState<IncidentCategoryValue | ''>('');
  const [severity, setSeverity] = useState<IncidentSeverity | ''>('');
  const [eventAtInput, setEventAtInput] = useState<string>(
    initialEventAt || new Date().toISOString().slice(0, 16).replace('T', ' ')
  );
  const [description, setDescription] = useState('');
  const [actionsTaken, setActionsTaken] = useState('');
  const [errors, setErrors] = useState<Record<string, string>>({});
  const [submitLock, setSubmitLock] = useState(false);

  const validateForm = (): {
    valid: boolean;
    parsedIso?: string;
  } => {
    const nextErrors: Record<string, string> = {};

    if (!category) {
      nextErrors.category = 'Select an incident category.';
    }
    if (!severity) {
      nextErrors.severity = 'Select a severity level.';
    }

    const trimmedDesc = description.trim();
    if (trimmedDesc.length < 20) {
      nextErrors.description =
        'Provide a factual narrative of at least 20 characters.';
    } else if (trimmedDesc.length > 3000) {
      nextErrors.description = 'Description must not exceed 3000 characters.';
    }

    const normalizedTimeStr = eventAtInput.trim().includes('T')
      ? eventAtInput.trim()
      : eventAtInput.trim().replace(' ', 'T');
    const parsedMs = Date.parse(normalizedTimeStr);

    if (Number.isNaN(parsedMs)) {
      nextErrors.eventAt =
        'Enter a valid date and time (e.g. 2026-10-19 11:15).';
    } else {
      const nowMs = Date.now();
      const fiveMinFuture = nowMs + 5 * 60 * 1000;
      const sevenDaysPast = nowMs - 7 * 24 * 60 * 60 * 1000;

      if (parsedMs > fiveMinFuture) {
        nextErrors.eventAt =
          'Incident time cannot be more than 5 minutes in the future.';
      } else if (parsedMs < sevenDaysPast) {
        nextErrors.eventAt = 'Incident time must be within the last 7 days.';
      } else if (shiftStartedAt) {
        const shiftStartMs = Date.parse(shiftStartedAt);
        if (!Number.isNaN(shiftStartMs) && parsedMs < shiftStartMs) {
          nextErrors.eventAt =
            'Incident time cannot be earlier than your active shift start time.';
        }
      }
    }

    setErrors(nextErrors);
    return {
      valid: Object.keys(nextErrors).length === 0,
      parsedIso: !Number.isNaN(parsedMs)
        ? new Date(parsedMs).toISOString()
        : undefined,
    };
  };

  const handlePressSave = async () => {
    if (isSubmitting || submitLock) return;
    const { valid, parsedIso } = validateForm();
    if (!valid || !category || !severity || !parsedIso) return;

    setSubmitLock(true);
    try {
      await onSubmitIncident({
        category,
        severity,
        eventAt: parsedIso,
        description: description.trim(),
        actionsTaken: actionsTaken.trim() ? actionsTaken.trim() : undefined,
      });
    } finally {
      setSubmitLock(false);
    }
  };

      behavior={Platform.OS === 'ios' ? 'padding' : undefined}

      <AppHeader
        title="Report Incident"
        subtitle={siteName}
        isOnline={isOnline}

      />

      >
        <EmergencyEscalationBanner />

        {storageError ? (
          <InfoAlertBanner
            tone="danger"
            title="Unable to Save Incident"
            body={storageError}
          />
        ) : null}

        <SurfaceCard>
          <View style={styles.siteContextRow}>
            <Text style={styles.siteContextLabel}>Assigned Site:</Text>
            <Text style={styles.siteContextValue}>{siteName}</Text>
          </View>

          <OptionPillGroup
            label="Incident Category"
            required
            options={INCIDENT_CATEGORY_OPTIONS}
            selectedValue={category}
            onSelect={(val) => {
              setCategory(val);
              if (errors.category) setErrors({ ...errors, category: '' });
            }}
            errorText={errors.category}
          />

          <OptionPillGroup
            label="Severity Level"
            required
            options={INCIDENT_SEVERITY_OPTIONS}
            selectedValue={severity}
            onSelect={(val) => {
              setSeverity(val);
              if (errors.severity) setErrors({ ...errors, severity: '' });
            }}
            errorText={errors.severity}
          />

          <FormField
            label="Time Incident Occurred"
            required
            helperText="YYYY-MM-DD HH:mm"
            value={eventAtInput}
            onChangeText={(txt) => {
              setEventAtInput(txt);
              if (errors.eventAt) setErrors({ ...errors, eventAt: '' });
            }}
            placeholder="2026-10-19 11:15"
            errorText={errors.eventAt}
          />

          <FormField
            label="What Happened? (Factual Narrative)"
            required
            helperText={`${description.trim().length}/3000 chars (min 20)`}
            value={description}
            onChangeText={(txt) => {
              setDescription(txt);
              if (errors.description) setErrors({ ...errors, description: '' });
            }}
            placeholder="Describe exactly what you observed, where on site it occurred, and any damage or persons involved without speculation."
            multiline
            maxLength={3000}
            errorText={errors.description}
          />

          <FormField
            label="Immediate Actions Taken"
            helperText="Optional"
            value={actionsTaken}
            onChangeText={setActionsTaken}
            placeholder="E.g. Secured gate, informed shift supervisor by telephone."
            multiline
            maxLength={1000}
          />
        </SurfaceCard>

        <View style={styles.actionStack}>

              isSubmitting || submitLock
                ? 'Saving to Encrypted Outbox...'
                : 'Save Incident Report'

            label="Cancel"

            variant="ghost"
            disabled={isSubmitting || submitLock}

};
